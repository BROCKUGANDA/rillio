    54→/* ---------------------------------- Reveal ---------------------------------- */
    55→
    56→function Reveal({
    57→  children,
    58→  delay = 0,
    59→  y = 28,
    60→  className,
    61→}: {
    62→  children: React.ReactNode;
    63→  delay?: number;
    64→  y?: number;
    65→  className?: string;
    66→}) {
    67→  const ref = useRef<HTMLDivElement>(null);
    68→  const inView = useInView(ref, { once: true, margin: "-80px" });
    69→  const reduce = useReducedMotion();
    70→  return (
    71→    <motion.div
    72→      ref={ref}
    73→      className={className}
    74→      initial={reduce ? { opacity: 0 } : { opacity: 0, y, filter: "blur(6px)" }}
    75→      animate={inView ? { opacity: 1, y: 0, filter: "blur(0px)" } : undefined}
    76→      transition={{ duration: 0.7, delay, ease: EASE }}
    77→    >
    78→      {children}
    79→    </motion.div>
    80→  );
    81→}

    83→/* --------------------------------- Counter --------------------------------- */

    85→function Counter({ to, prefix = "", suffix = "", decimals = 0 }: { to: number; prefix?: string; suffix?: string; decimals?: number }) {
    86→  const ref = useRef<HTMLSpanElement>(null);
    87→  const inView = useInView(ref, { once: true });
    88→  const [val, setVal] = useState(0);
    89→  useEffect(() => {
    90→    if (!inView) return;
    91→    const controls = animate(0, to, {
    92→      duration: 1.6,
    93→      ease: EASE,
    94→      onUpdate: (v) => setVal(v),
    95→    });
    96→    return () => controls.stop();
    97→  }, [inView, to]);
    98→  return (
    99→    <span ref={ref} className="num">
   100→      {prefix}
   101→      {val.toFixed(decimals)}
   102→      {suffix}
   103→    </span>
   104→  );
   105→}
